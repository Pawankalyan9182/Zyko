const CATEGORIES = [
  "All",
  "Fruits & Vegetables",
  "Grocery & Kitchen",
  "Dairy & Bread",
  "Snacks & Drinks",
  "Beauty & Personal Care",
  "Household Essentials",
];

// ================= APP =================

export default function App() {
  // ---------- LOGIN ----------
  const [user, setUser] = useState(null);
  const [loginStep, setLoginStep] = useState("mobile");

  const [mobile, setMobile] = useState("");
  const [otp, setOtp] = useState("");

  const [profileName, setProfileName] = useState("");
  const [profileAddress, setProfileAddress] = useState("");

  // ---------- EDIT PROFILE ----------
  const [editName, setEditName] = useState("");
  const [editAddress, setEditAddress] = useState("");

  // ---------- APP ----------
  const [screen, setScreen] = useState("home");

  const [cart, setCart] = useState({});
  const [searchText, setSearchText] = useState("");
  const [selectedCategory, setSelectedCategory] = useState("All");

  const [paymentMethod, setPaymentMethod] = useState("UPI");

  const [orderStatus, setOrderStatus] = useState("confirmed");
  const [orderNumber, setOrderNumber] = useState("");

  const [orders, setOrders] = useState([]);

  // ================= TRACKING =================

  useEffect(() => {
    if (screen !== "tracking") return;

    if (orderStatus !== "confirmed") return;

    const packingTimer = setTimeout(() => {
      setOrderStatus("packing");
    }, 3000);

    const deliveryTimer = setTimeout(() => {
      setOrderStatus("out_for_delivery");
    }, 8000);

    const deliveredTimer = setTimeout(() => {
      setOrderStatus("delivered");
    }, 13000);

    return () => {
      clearTimeout(packingTimer);
      clearTimeout(deliveryTimer);
      clearTimeout(deliveredTimer);
    };
  }, [screen, orderStatus]);

  // ================= CART =================

  const addToCart = (product) => {
    setCart((prev) => ({
      ...prev,
      [product.id]: (prev[product.id] || 0) + 1,
    }));
  };

  const increase = (product) => {
    addToCart(product);
  };

  const decrease = (product) => {
    setCart((prev) => {
      const updated = { ...prev };

      if (updated[product.id] > 1) {
        updated[product.id]--;
      } else {
        delete updated[product.id];
      }

      return updated;
    });
  };

  const cartItems = PRODUCTS.filter((p) => cart[p.id]);

  const subtotal = cartItems.reduce(
    (sum, product) => sum + product.price * cart[product.id],
    0
  );

  const deliveryFee = subtotal > 299 || subtotal === 0 ? 0 : 20;

  const total = subtotal + deliveryFee;

  const cartCount = cartItems.reduce(
    (sum, product) => sum + cart[product.id],
    0
  );

  // ================= PRODUCTS FILTER =================

  const filteredProducts = PRODUCTS.filter((product) => {
    const categoryMatch =
      selectedCategory === "All" ||
      product.category === selectedCategory;

    const searchMatch = product.name
      .toLowerCase()
      .includes(searchText.toLowerCase());

    return categoryMatch && searchMatch;
  });

  // ================= LOGIN =================

  const sendOTP = () => {
    if (!/^[0-9]{10}$/.test(mobile)) {
      Alert.alert(
        "Invalid Mobile Number",
        "Please enter a valid 10 digit mobile number."
      );
      return;
    }

    setLoginStep("otp");

    Alert.alert(
      "Demo OTP",
      "Your demo OTP is 1234.\nReal SMS OTP will be connected later."
    );
  };

  const verifyOTP = () => {
    if (otp !== "1234") {
      Alert.alert("Wrong OTP", "For demo use OTP: 1234");
      return;
    }

    setLoginStep("profile");
  };

  const saveProfile = () => {
    if (!profileName.trim()) {
      Alert.alert("Enter Name", "Please enter your name.");
      return;
    }

    if (!profileAddress.trim()) {
      Alert.alert("Enter Address", "Please enter your delivery address.");
      return;
    }

    setUser({
      mobile,
      name: profileName.trim(),
      address: profileAddress.trim(),
    });

    setScreen("home");
  };

  // ================= EDIT PROFILE =================

  const openEditProfile = () => {
    setEditName(user?.name || "");
    setEditAddress(user?.address || "");
    setScreen("editProfile");
  };

  const saveEditedProfile = () => {
    if (!editName.trim() || !editAddress.trim()) {
      Alert.alert("Missing Information", "Please fill all details.");
      return;
    }

    setUser({
      ...user,
      name: editName.trim(),
      address: editAddress.trim(),
    });

    Alert.alert("Success", "Profile updated successfully.");
    setScreen("profile");
  };

  // ================= ORDER =================

  const placeOrder = () => {
    if (!user) {
      setLoginStep("mobile");
      setScreen("home");
      return;
    }

    if (cartItems.length === 0) {
      Alert.alert("Empty Cart", "Please add products first.");
      return;
    }

    const newOrderNumber =
      "TJ" + Math.floor(100000 + Math.random() * 900000);

    const newOrder = {
      id: newOrderNumber,
      items: cartItems.map((p) => ({
        ...p,
        quantity: cart[p.id],
      })),
      total,
      payment: paymentMethod,
      address: user.address,
      status: "confirmed",
      date: new Date().toLocaleDateString(),
    };

    setOrders((prev) => [newOrder, ...prev]);

    setOrderNumber(newOrderNumber);
    setOrderStatus("confirmed");

    setScreen("tracking");
  };

  // ================= LOGIN UI =================

  if (!user) {
    return (
      <SafeAreaView style={styles.loginContainer}>
        <ScrollView contentContainerStyle={styles.loginContent}>
          <Text style={styles.logo}>Tejovanam</Text>

          <Text style={styles.deliveryText}>
            Groceries delivered in 10 minutes ⚡
          </Text>

          {loginStep === "mobile" && (
            <View style={styles.loginBox}>
              <Text style={styles.loginTitle}>Welcome 👋</Text>

              <Text style={styles.loginSubtitle}>
                Login with your mobile number
              </Text>

              <TextInput
                style={styles.input}
                placeholder="Enter 10 digit mobile number"
                keyboardType="phone-pad"
                maxLength={10}
                value={mobile}
                onChangeText={setMobile}
              />

              <TouchableOpacity
                style={styles.primaryButton}
                onPress={sendOTP}
              >
                <Text style={styles.primaryButtonText}>
                  SEND OTP
                </Text>
              </TouchableOpacity>
            </View>
          )}

          {loginStep === "otp" && (
            <View style={styles.loginBox}>
              <Text style={styles.loginTitle}>Verify OTP 🔐</Text>

              <Text style={styles.loginSubtitle}>
                OTP sent to +91 {mobile}
              </Text>

              <TextInput
                style={styles.input}
                placeholder="Enter OTP"
                keyboardType="number-pad"
                maxLength={4}
                value={otp}
                onChangeText={setOtp}
              />

              <TouchableOpacity
                style={styles.primaryButton}
                onPress={verifyOTP}
              >
                <Text style={styles.primaryButtonText}>
                  VERIFY OTP
                </Text>
              </TouchableOpacity>

              <TouchableOpacity
                onPress={() => setLoginStep("mobile")}
              >
                <Text style={styles.linkText}>
                  Change mobile number
                </Text>
              </TouchableOpacity>
            </View>
          )}

          {loginStep === "profile" && (
            <View style={styles.loginBox}>
              <Text style={styles.loginTitle}>
                Complete Profile 👤
              </Text>

              <TextInput
                style={styles.input}
                placeholder="Your name"
                value={profileName}
                onChangeText={setProfileName}
              />

              <TextInput
                style={[styles.input, styles.addressInput]}
                placeholder="Delivery address"
                multiline
                value={profileAddress}
                onChangeText={setProfileAddress}
              />

              <TouchableOpacity
                style={styles.primaryButton}
                onPress={saveProfile}
              >
                <Text style={styles.primaryButtonText}>
                  SAVE & CONTINUE
                </Text>
              </TouchableOpacity>
            </View>
          )}
        </ScrollView>
      </SafeAreaView>
    );
  }

  // ================= HOME =================

  if (screen === "home") {
    return (
      <SafeAreaView style={styles.container}>
        <ScrollView>
          <View style={styles.header}>
            <View>
              <Text style={styles.brand}>Tejovanam ⚡</Text>
              <Text style={styles.location}>
                📍 {user.address}
              </Text>
            </View>

            <TouchableOpacity
              style={styles.profileCircle}
              onPress={() => setScreen("profile")}
            >
              <Text>👤</Text>
            </TouchableOpacity>
          </View>

          <Text style={styles.greeting}>
            Hi, {user.name} 👋
          </Text>

          <TextInput
            style={styles.search}
            placeholder="Search groceries, fruits & more"
            value={searchText}
            onChangeText={(text) => {
              setSearchText(text);
              setScreen("products");
            }}
          />

          <View style={styles.banner}>
            <Text style={styles.bannerTitle}>
              Groceries in 10 minutes ⚡
            </Text>

            <Text style={styles.bannerSubtitle}>
              Fresh products delivered fast
            </Text>
          </View>

          <Text style={styles.sectionTitle}>
            Shop by Category
          </Text>

          <ScrollView
            horizontal
            showsHorizontalScrollIndicator={false}
          >
            {CATEGORIES.slice(1).map((category) => (
              <TouchableOpacity
                key={category}
                style={styles.categoryCard}
                onPress={() => {
                  setSelectedCategory(category);
                  setScreen("products");
                }}
              >
                <Text style={styles.categoryEmoji}>
                  {category === "Fruits & Vegetables"
                    ? "🥦"
                    : category === "Grocery & Kitchen"
                    ? "🍚"
                    : category === "Dairy & Bread"
                    ? "🥛"
                    : category === "Snacks & Drinks"
                    ? "🍿"
                    : category === "Beauty & Personal Care"
                    ? "🧴"
                    : "🧹"}
                </Text>

                <Text style={styles.categoryText}>
                  {category}
                </Text>
              </TouchableOpacity>
            ))}
          </ScrollView>

          <Text style={styles.sectionTitle}>
            Frequently Bought
          </Text>

          {PRODUCTS.slice(0, 6).map((product) => (
            <ProductCard
              key={product.id}
              product={product}
              quantity={cart[product.id] || 0}
              onAdd={addToCart}
              onIncrease={increase}
              onDecrease={decrease}
            />
          ))}
        </ScrollView>

        <BottomNav
          screen={screen}
          setScreen={setScreen}
          cartCount={cartCount}
        />
      </SafeAreaView>
    );
  }

  // ================= CATEGORIES =================

  if (screen === "categories") {
    return (
      <SafeAreaView style={styles.container}>
        <Text style={styles.pageTitle}>Categories</Text>

        <ScrollView>
          {CATEGORIES.slice(1).map((category) => (
            <TouchableOpacity
              key={category}
              style={styles.bigCategory}
              onPress={() => {
                setSelectedCategory(category);
                setSearchText("");
                setScreen("products");
              }}
            >
              <Text style={styles.bigCategoryEmoji}>
                {category === "Fruits & Vegetables"
                  ? "🥦"
                  : category === "Grocery & Kitchen"
                  ? "🍚"
                  : category === "Dairy & Bread"
                  ? "🥛"
                  : category === "Snacks & Drinks"
                  ? "🍿"
                  : category === "Beauty & Personal Care"
                  ? "🧴"
                  : "🧹"}
              </Text>

              <Text style={styles.bigCategoryText}>
                {category}
              </Text>

              <Text>›</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>

        <BottomNav
          screen={screen}
          setScreen={setScreen}
          cartCount={cartCount}
        />
      </SafeAreaView>
    );
  }

  // ================= PRODUCTS =================

  if (screen === "products") {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.productHeader}>
          <TouchableOpacity onPress={() => setScreen("home")}>
            <Text style={styles.back}>‹</Text>
          </TouchableOpacity>

          <Text style={styles.pageTitle}>
            {selectedCategory}
          </Text>
        </View>

        <TextInput
          style={styles.search}
          placeholder="Search products"
          value={searchText}
          onChangeText={setSearchText}
        />

        <ScrollView>
          {filteredProducts.length === 0 ? (
            <Text style={styles.emptyText}>
              No products found
            </Text>
          ) : (
            filteredProducts.map((product) => (
              <ProductCard
                key={product.id}
                product={product}
                quantity={cart[product.id] || 0}
                onAdd={addToCart}
                onIncrease={increase}
                onDecrease={decrease}
              />
            ))
          )}
        </ScrollView>

        {cartCount > 0 && (
          <TouchableOpacity
            style={styles.viewCart}
            onPress={() => setScreen("cart")}
          >
            <Text style={styles.viewCartText}>
              View Cart • ₹{total}
            </Text>
          </TouchableOpacity>
        )}

        <BottomNav
          screen={screen}
          setScreen={setScreen}
          cartCount={cartCount}
        />
      </SafeAreaView>
    );
  }

  // ================= CART =================

  if (screen === "cart") {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.productHeader}>
          <TouchableOpacity onPress={() => setScreen("home")}>
            <Text style={styles.back}>‹</Text>
          </TouchableOpacity>

          <Text style={styles.pageTitle}>Your Cart</Text>
        </View>

        <ScrollView>
          {cartItems.map((product) => (
            <View key={product.id} style={styles.cartItem}>
              <Image
                source={{ uri: product.image }}
                style={styles.cartImage}
              />

              <View style={{ flex: 1 }}>
                <Text style={styles.productName}>
                  {product.name}
                </Text>

                <Text style={styles.productUnit}>
                  {product.unit}
                </Text>

                <Text style={styles.productPrice}>
                  ₹{product.price}
                </Text>
              </View>

              <View style={styles.quantityBox}>
                <TouchableOpacity
                  onPress={() => decrease(product)}
                >
                  <Text style={styles.quantityButton}>−</Text>
                </TouchableOpacity>

                <Text style={styles.quantity}>
                  {cart[product.id]}
                </Text>

                <TouchableOpacity
                  onPress={() => increase(product)}
                >
                  <Text style={styles.quantityButton}>+</Text>
                </TouchableOpacity>
              </View>
            </View>
          ))}

          {cartItems.length === 0 && (
            <Text style={styles.emptyText}>
              Your cart is empty 🛒
            </Text>
          )}

          {cartItems.length > 0 && (
            <View style={styles.bill}>
              <Text style={styles.sectionTitle}>
                Bill Details
              </Text>

              <Row
                title="Subtotal"
                value={`₹${subtotal}`}
              />

              <Row
                title="Delivery Fee"
                value={
                  deliveryFee === 0
                    ? "FREE"
                    : `₹${deliveryFee}`
                }
              />

              <View style={styles.line} />

              <Row
                title="Total"
                value={`₹${total}`}
                bold
              />

              <TouchableOpacity
                style={styles.primaryButton}
                onPress={() => setScreen("checkout")}
              >
                <Text style={styles.primaryButtonText}>
                  PROCEED TO CHECKOUT
                </Text>
              </TouchableOpacity>
            </View>
          )}
        </ScrollView>

        <BottomNav
          screen={screen}
          setScreen={setScreen}
          cartCount={cartCount}
        />
      </SafeAreaView>
    );
  }

  // ================= CHECKOUT =================

  if (screen === "checkout") {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.productHeader}>
          <TouchableOpacity onPress={() => setScreen("cart")}>
            <Text style={styles.back}>‹</Text>
          </TouchableOpacity>

          <Text style={styles.pageTitle}>Checkout</Text>
        </View>

        <ScrollView>
          <View style={styles.checkoutBox}>
            <Text style={styles.sectionTitle}>
              ⚡ Delivery
            </Text>

            <Text style={styles.deliveryTime}>
              Delivery in 10 minutes
            </Text>

            <Text style={styles.addressTitle}>
              📍 Deliver to
            </Text>

            <Text style={styles.addressText}>
              {user.address}
            </Text>

            <TouchableOpacity
              onPress={openEditProfile}
            >
              <Text style={styles.linkText}>
                Change address
              </Text>
            </TouchableOpacity>
          </View>

          <View style={styles.checkoutBox}>
            <Text style={styles.sectionTitle}>
              Payment Method
            </Text>

            {["UPI", "Card", "COD"].map((method) => (
              <TouchableOpacity
                key={method}
                style={[
                  styles.paymentOption,
                  paymentMethod === method &&
                    styles.selectedPayment,
                ]}
                onPress={() => setPaymentMethod(method)}
              >
                <Text style={styles.paymentText}>
                  {method === "UPI"
                    ? "📱 UPI"
                    : method === "Card"
                    ? "💳 Card"
                    : "💵 Cash on Delivery"}
                </Text>

                <Text>
                  {paymentMethod === method ? "✓" : ""}
                </Text>
              </TouchableOpacity>
            ))}
          </View>

          <View style={styles.checkoutBox}>
            <Text style={styles.sectionTitle}>
              Order Summary
            </Text>

            <Row
              title="Subtotal"
              value={`₹${subtotal}`}
            />

            <Row
              title="Delivery"
              value={
                deliveryFee === 0
                  ? "FREE"
                  : `₹${deliveryFee}`
              }
            />

            <View style={styles.line} />

            <Row
              title="Total"
              value={`₹${total}`}
              bold
            />

            <TouchableOpacity
              style={styles.primaryButton}
              onPress={placeOrder}
            >
              <Text style={styles.primaryButtonText}>
                PLACE ORDER • ₹{total}
              </Text>
            </TouchableOpacity>
          </View>
        </ScrollView>
      </SafeAreaView>
    );
  }

  // ================= TRACKING =================

  if (screen === "tracking") {
    return (
      <SafeAreaView style={styles.container}>
        <ScrollView>
          <Text style={styles.trackingTitle}>
            Order Tracking 🚴
          </Text>

          <View style={styles.trackingCard}>
            <Text style={styles.orderNumber}>
              Order #{orderNumber}
            </Text>

            <Text style={styles.bigTen}>
              10
            </Text>

            <Text style={styles.minutes}>
              MINUTES
            </Text>

            <Text style={styles.statusText}>
              {orderStatus === "confirmed"
                ? "Order Confirmed ✓"
                : orderStatus === "packing"
                ? "Packing your order 📦"
                : orderStatus === "out_for_delivery"
                ? "Out for Delivery 🚴"
                : "Delivered Successfully 🎉"}
            </Text>
          </View>

          <View style={styles.timeline}>
            <StatusRow
              title="Order Confirmed"
              active={true}
            />

            <StatusRow
              title="Packing"
              active={[
                "packing",
                "out_for_delivery",
                "delivered",
              ].includes(orderStatus)}
            />

            <StatusRow
              title="Out for Delivery"
              active={[
                "out_for_delivery",
                "delivered",
              ].includes(orderStatus)}
            />

            <StatusRow
              title="Delivered"
              active={orderStatus === "delivered"}
            />
          </View>

          <View style={styles.checkoutBox}>
            <Text style={styles.sectionTitle}>
              Delivery Address
            </Text>
